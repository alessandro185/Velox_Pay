Exemplo em JavaScript
fetch 
const response = await fetch("http://127.0.0.1:8000/pagamentos", {
  method: "POST",
  headers: { "Content-Type": "application/json" },
  body: JSON.stringify({
    id_pedido: 101,
    valor_total: 49.90,
    metodo_pagamento: "Pix"
  })
});

const pagamento = await response.json();
Exemplo em Python
requests 
import requests

payload = {
    "id_pedido": 101,
    "valor_total": 49.90,
    "metodo_pagamento": "Pix"
}

res = requests.post(
    "http://127.0.0.1:8000/pagamentos",
    json=payload
)

print(res.json())
